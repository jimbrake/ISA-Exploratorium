
	Various constants for configuring memory and pointer sizes
inst_size         9   instructions are either 9 or 18 bits
prog_adr_size     8   2024x9 program memory max
prog_size       256   2024x9 program memory max
bitRAM_adr_size   5   data memory address size, max is 5
bitRAM_size      32   32X1 data RAM max
rtn_stack_size	  8   32x11 return stack memory max
rtn_ptr_size	  3   return stack memory pointer

	One word Op-codes
         XXXRPSSUU
op_LD    000000000 mem(SS,UU) => A, push DS if P=1
op_LDC   001000000 not mem(SS,UU) => A, push DS if P=1
op_AND   010000000 A and mem(SS,UU) => A, push DS if P=1
op_OR    011000000 A or mem(SS,UU) => A, push DS if P=1
op_XOR   100000000 A xor mem(SS,UU) => A, push DS if P=1
op_ADC   101000000 A + mem(SS,UU) + C => A, push DS if P=1, update C
op_ST    000100000 A => mem(SS,UU), pop DS if P=1
op_STC   001100000 not A => mem(SS,UU), pop DS if P=1
op_RAND  010100000 A and mem(SS,UU) => mem(SS,UU), pop DS if P=1
op_ROR   011100000 A or mem(SS,UU) => mem(SS,UU), pop DS if P=1
op_RXOR  100100000 A xor mem(SS,UU) => mem(SS,UU), pop DS if P=1
op_RADC  101100000 A + mem(SS,UU) + C => mem(SS,UU), pop DS if P=1, update C
op_INCM  110000000 mem(SS,UU) + C => mem(SS,UU), update C
op_DECM  110100000 '1' + mem(SS,UU) + C => mem(SS,UU), update C
op_MACS  111000000 S & CC & AA (S:swap, CC & AA: nop,complement,0,1)

	Two word Op-codes
         1111XXXNN NNNNNNNN
op_CALL  111100000 N => PC, RP - 1 => RP, PC+1 => mem(RP) 
op_JMP   111100100 N => PC
op_JAC   111101000 N => PC if A = 0, else PC + 1 => PC
op_JAS   111101100 N => PC if A = 1, else PC + 1 => PC
op_JCC   111110000 N => PC if C = 0, else PC + 1 => PC
op_JCS   111110100 N => PC if C = 1, else PC + 1 => PC

    MACS Field modifiers
		 1110SCCAA		Can S, CC & AA be OR'd together?
op_RTN   111000000 mem(RP) => PC, RP + 1 => RP (replaces a null inst)
op_SWAP  111010000 swap A and C, done after modifications to A & C
op_NOTA  111000001 complement A
op_CLRA  111000010 set A to zero
op_SETA  111000011 set A to one
op_NOTC  111000100 complement C
op_CLRC  111001000 set C to zero
op_SETC  111001100 set C to one
op_XXX   111011X1X redundant, can use for four additional op-codes
